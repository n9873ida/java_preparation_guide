    private static void defaultFI() {
        Supplier<String> supplier = () -> "hello from supplier";
        System.out.println(supplier.get());

        Consumer<String> consumer = System.out::println;
        consumer.accept("hello from consumer");

        Predicate<String> predicate = (str) -> str.startsWith("hello") && str.endsWith("predicate");
        System.out.println(predicate.test("hello from predicate"));
        System.out.println(predicate.test("good bye from predicate"));

        Function<String, Integer> function = String::length;
        System.out.println(function.apply("hello from function FI"));

        BiFunction<Integer, String, Integer> biFunction = (val, str2) -> val + str2.length();
        System.out.println(biFunction.apply(15, "from bifunction"));

        UnaryOperator<String> uOp = (str) -> String.valueOf(str.length());
        System.out.println(uOp.apply("hello from unary operator"));

        BinaryOperator<String> bOp = (str1, str2) -> str1 + str2;
        System.out.println(bOp.apply("Hello", "BiFunction"));
    }

    private static int binarySearch(int elem) {
        int[] num = {1, 4, 5, 7, 8, 10, 15, 18, 29, 48, 85, 86, 99};

        int l = 0, r = num.length - 1;

        while (l <= r) {
            int mid = l + (r - l) / 2;
            if (num[mid] == elem) return mid;
            if (num[mid] < elem) {
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
        return -1;
    }

    private static void stringCount() {
        String str = "XXXSSXXSXXXVVSSQQAIIRMTTXXX";

        Map<String, Integer> map = new HashMap<>();

        int i = 0;
        while (i < str.length()) {
            int j = i + 1;
            while (j < str.length() && str.charAt(j - 1) == str.charAt(j)) {
                j++;
            }
            String sub = str.substring(i, j);
            i = j;
            map.put(sub, map.getOrDefault(sub, 0) + 1);
        }

        map.entrySet().stream().sorted(Comparator.comparing(Map.Entry<String, Integer>::getValue, Comparator.reverseOrder()).thenComparing(Map.Entry::getKey, Comparator.reverseOrder())).forEach(System.out::println);

    private static void topKElements() {
        // 1. Build frequency map
        int[] nums = {1, 2, 4, 4, 4, 1, 5, 1, 2, 1};
        int k = 3;
        Map<Integer, Integer> freq = new HashMap<>();
        for (int n : nums) {
            freq.put(n, freq.getOrDefault(n, 0) + 1);
        }

        // 2. Min‐heap ordered by frequency
        PriorityQueue<Map.Entry<Integer, Integer>> pq =
                new PriorityQueue<>(Comparator.comparingInt(Map.Entry::getValue));

        // 3. Keep only top k entries in the heap
        for (Map.Entry<Integer, Integer> e : freq.entrySet()) {
            pq.offer(e);
            if (pq.size() > k) {
                pq.poll();
            }
        }

        // 4. Extract keys into result array (reverse order yields highest freq first)
        int[] res = new int[k];
        for (int i = k - 1; i >= 0; i--) {
            res[i] = pq.poll().getKey();
        }
        for (int re : res) {
            System.out.println(re);
        }
    }

    // find prime numbers between 2 and 100 which does not contain 3 and 7
    private static void findPrimeNumber() {

        List<Integer> primes = new ArrayList<>();
        for (int i = 2; i <= 100; i++) {

            int digit = i;
            boolean bad = false;
            while (digit > 0) {
                if (digit % 10 == 3 || digit % 10 == 7) {
                    bad = true;
                    break;
                }
                digit /= 10;
            }
            if (bad == true) continue;

            for (int j = 2; j * j <= i; j++) {
                if (i % j == 0) {
                    continue;
                }
            }
            primes.add(i);
        }

        primes.forEach(x -> System.out.print(x + " "));
    }

    private static void separateOneSide() {
        int[] arr = new int[]{5, 2, 4, 1, 2, 3, 6, 45, 1, 1, 4, 2, 1, 2, 4, 1};

        int[] arr2 = {1, 2, 4, 5, 3, 5, 4, 1, 6, 1, 1, 1, 1, 3};
        Arrays.stream(arr2).filter(x -> x != 1).boxed()
                .collect(Collectors.groupingBy(x -> x, Collectors.counting())).entrySet().stream().forEach(System.out::println);
        int update = 0;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == 1) {
                int temp = arr[update];
                arr[update++] = arr[i];
                arr[i] = temp;
            }
        }

        int pos = arr.length - 1;
        for (int i = arr.length - 1; i >= 0; i--) {
            if (arr[i] != 1) {
                int temp = arr[i];
                arr[i] = arr[pos];
                arr[pos--] = temp;
            }
        }

        while (pos >= 0) {
            arr[pos--] = 1;
        }

        for (int num : arr) {
            System.out.print(num + ", ");

class Employee {
    private String department;
    private String name;
    private Integer salary;
    private String city;

    public Employee(String department, String name, Integer salary, String city) {
        this.department = department;
        this.name = name;
        this.salary = salary;
        this.city = city;
    }

    public String getDepartment() {
        return this.department;
    }

    public String getName() {
        return this.name;
    }

    public Integer getSalary() {
        return this.salary;
    }

    public String getCity() {
        return this.city;
    }

    @Override
    public String toString() {
        return "department: " + this.department + ", name: " + name + ", salary: " + salary + ", city: " + city;
    }
}
